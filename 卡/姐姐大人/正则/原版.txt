      // 欄位名：簡繁、別名都認；長的寫前面（角色卡名 要比 角色卡 先比對）
      const 欄位 = [
        ['名', ['角色卡名']],
        ['tag', ['tag', 'tags', '標籤', '标签']],
        ['信息', ['角色信息', '角色資訊', '角色資料']],
        ['簡介', ['作品简介', '作品簡介']],

        ['作者', ['作者']],
        ['角色卡', ['角色卡']],
        ['性取向', ['性取向']],
        ['二傳', ['二傳', '二传']],
        ['二改', ['二改']],
        ['商用', ['商用', '是否允许商业平台使用', '是否允許商業平台使用']],
        ['腳本', ['腳本', '脚本']],
        ['模板', ['提示詞模板是否需要安裝', '提示词模板是否需要安装']],
        ['小手機', ['小手機', '小手机']],
        ['其他軟件', ['是否可用在其他軟件', '是否可用在其他软件']],
        ['性別限制', ['user性別限制', 'user性别限制']],
      ];
      const 顯示名 = {
        作者: '作者',
        角色卡: '角色卡',
        性取向: '性取向',
        二傳: '二傳',

        商用: '商業平台使用',

        模板: '提示詞模板',
        小手機: '小手機',
        其他軟件: '其他軟體使用',
        性別限制: 'user 性別限制',
      };

      const 別名表 = [];
      for (const [key, names] of 欄位) for (const n of names) 別名表.push([n, key]);
      別名表.sort((a, b) => b[0].length - a[0].length);

      function 拆欄位(text) {
        const out = {};
        let 目前 = null;
        for (const line of text.split(/\r?\n/)) {
          const t = line.trim();
          const hit = 別名表.find(([n]) => t.toLowerCase().startsWith(n.toLowerCase()) && /^[：:]/.test(t.slice(n.length).trim()));
          if (hit) {
            目前 = hit[1];
            out[目前] = t.slice(hit[0].length).trim().replace(/^[：:]\s*/, '');
          } else if (目前) {
            out[目前] += '\n' + line;
          }
        }
        for (const k in out) out[k] = out[k].trim();
        return out;
      }

      function 純文字(html) {
        const box = document.createElement('div');
        box.innerHTML = html;
        return box;
      }

      // 角色信息：<p><span style="font-size…">名字</span></p> 開頭的是一個人，後面的段落都算他的介紹
      function 拆人物(html) {
        const box = 純文字(html);
        const ps = box.querySelectorAll('p');
        if (!ps.length) return [{ 名: '', 段: [box.textContent.trim()] }];
        const 人 = [];
        ps.forEach(p => {
          const span = p.querySelector('span[style*="font-size"]');
          if (span) {
            const 名 = span.textContent.trim();
            span.remove();
            人.push({ 名, 段: [] });
          }
          const 文 = p.textContent.trim();
          if (!文) return;
          if (!人.length) 人.push({ 名: '', 段: [] });
          人[人.length - 1].段.push(文);
        });
        return 人;
      }

      function 加(parent, tag, text, cls) {
        const el = document.createElement(tag);
        if (text !== undefined) el.textContent = text;
        if (cls) el.className = cls;
        parent.appendChild(el);
        return el;
      }

      function 套主题() {
        try {
          const root = getComputedStyle(window.parent.document.documentElement);
          const 取 = name => root.getPropertyValue(name).trim();
          const set = (k, v) => v && document.documentElement.style.setProperty(k, v);
          set('--fg', 取('--SmartThemeBodyColor'));
          set('--bd', 取('--SmartThemeBorderColor'));
          set('--bg', 取('--SmartThemeBlurTintColor'));
          set('--font', getComputedStyle(window.parent.document.body).fontFamily);
        } catch (e) {
          // 拿不到父頁面就維持預設
        }
      }

      const 資料 = 拆欄位(document.getElementById('raw').innerHTML);

      document.getElementById('name').textContent = 資料.名 || '';
      document.getElementById('motto').textContent = 資料.節選 || '';
      if (!資料.節選) document.getElementById('motto').hidden = true;

      const tags = document.getElementById('tags');
      String(資料.tag || '')
        .split(/[、,，/／]/)
        .map(s => s.trim())
        .filter(Boolean)
        .forEach(s => 加(tags, 'span', s, 'tag'));

      const duo = document.getElementById('duo');
      拆人物(資料.信息 || '').forEach(人 => {
        const box = 加(duo, 'div', undefined, 'persona');
        if (人.名) 加(box, 'h2', 人.名);
        人.段.forEach(s => 加(box, 'p', s));
      });
      if (duo.children.length < 2) duo.style.gridTemplateColumns = '1fr';

      const intro = document.getElementById('intro');
      if (資料.簡介) {
        let 空行 = false;
        純文字(資料.簡介)
          .textContent.split(/\r?\n/)
          .forEach(line => {
            const t = line.trim();
            if (!t) return void (空行 = true);
            加(intro, 'p', t, 空行 ? 'gap' : '');
            空行 = false;
          });
      } else {
        document.getElementById('intro-wrap').hidden = true;
      }

      const dl = document.getElementById('details');
      for (const key in 顯示名) {
        if (!資料[key]) continue;
        加(加(dl, 'dt'), 'span', 顯示名[key]);
        加(dl, 'dd', 純文字(資料[key]).textContent.trim());
      }

      const page1 = document.getElementById('page1');
      const page2 = document.getElementById('page2');
      document.getElementById('to2').addEventListener('click', () => {
        page1.hidden = true;
        page2.hidden = false;
      });
      document.getElementById('to1').addEventListener('click', () => {
        page2.hidden = true;
        page1.hidden = false;
      });

      套主题();
    </script>
