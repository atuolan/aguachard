    <script>
      const 场合名 = { 职场: '職場', 私下: '私下', 营业: '營業' };
      const 旧模式 = { 上司: '職場', 伪娘: '營業', 私下状态: '私下', 恋人: '私下' };

      function 设(k, text) {
        document.querySelectorAll('[data-k="' + k + '"]').forEach(el => (el.textContent = text));
      }

      function 读() {
        const id = getCurrentMessageId();
        const vars = getVariables({ type: 'message', message_id: id });
        return vars && vars.stat_data && vars.stat_data.黎靖青 ? vars.stat_data : null;
      }

      let 上次 = '';
      function 画() {
        let stat = null;
        try {
          stat = 读();
        } catch (e) {
          stat = null;
        }
        const key = JSON.stringify(stat);
        if (key === 上次) return;
        上次 = key;
        const panel = document.getElementById('panel');
        if (!stat) {
          panel.hidden = true;
          return;
        }
        panel.hidden = false;

        const 人 = stat.黎靖青 || {};
        const 关系 = stat.关系 || {};
        const 状态 = stat._状态 || {};
        const 试 = 状态.身分试探 || {};
        const 穷 = 人.目前身份 === '穷云海';
        panel.classList.toggle('mode-maid', 穷);
        panel.classList.toggle('mode-boss', !穷);

        设('时间点', (stat.世界 && stat.世界.时间点) || '');
        const 场合 = 场合名[人.场合] || 旧模式[人.目前模式] || '';
        设('身份场合', (穷 ? '窮雲海' : '黎靖青') + (场合 ? '・' + 场合 : ''));
        const 揭穿 =
          stat.user && stat.user.识破身份 === '识破' && stat['黎靖青/穷云海'] && stat['黎靖青/穷云海'].认出聊天对象 === '认出';
        设('关系', '現實：' + (关系.现实 || '—') + (揭穿 ? '' : '　線上：' + (关系.线上 || '—')));

        const 识破 = stat.user && stat.user.识破身份 === '识破';
        const 认出 = stat['黎靖青/穷云海'] && stat['黎靖青/穷云海'].认出聊天对象 === '认出';
        设('识破', 识破 ? '你已識破他' : '你未識破他');
        设('认出', 认出 ? '他已認出你' : '他未認出你');
        document.querySelector('[data-k="识破"]').classList.toggle('on', 识破);
        document.querySelector('[data-k="认出"]').classList.toggle('on', 认出);

        let 事件 = '';
        if (试.进行中) 事件 = '身分試探進行中（' + Math.min(Number(试.进度) || 0, 20) + '／20）';
        else if (试.结果 === '接受') 事件 = '身分試探：你接受了完整的他';
        else if (试.结果 === '抗拒') 事件 = '身分試探：你沒有接受他';
        else if (试.结果 === '回避') 事件 = '身分試探：你迴避了';
        设('事件', 事件);

        设('心情', 人.心情 || '');
        const 冷 = Number(人.冷淡值) || 0;
        const 亲 = Number(人.亲密值) || 0;
        const 兴 = Number(人.兴奋值) || 0;
        设('冷淡值', '冷淡值：' + 冷);
        设('亲密值', '親密值：' + 亲);
        设('兴奋值', '興奮值：' + 兴);
        const 锁 = 人.性爱场景中 ? '親密中，暫不變動' : 试.进行中 ? '試探中，暫不變動' : '';
        设('冷淡备注', 锁 || (穷 ? '窮雲海身分時不變動' : ''));
        设('亲密备注', 锁);
        document.querySelector('.cold-progress').style.width = 冷 + '%';
        document.querySelector('.intimacy-progress').style.width = 亲 + '%';
        document.querySelector('.excitement-progress').style.width = 兴 + '%';

        设('内心想法', 人.内心想法 || '');
        const list = document.querySelector('.todo-list');
        list.innerHTML = '';
        String(人.待办事项 || '')
          .split('|')
          .map(s => s.trim())
          .filter(Boolean)
          .forEach(item => {
            const div = document.createElement('div');
            div.className = 'todo-item';
            div.textContent = item;
            list.appendChild(div);
          });
      }

      // 模型更新變量、swipe、重新處理變量都會改到樓層變量；輪詢比逐一掛事件穩
      画();
      setInterval(画, 2000);
    </script>
