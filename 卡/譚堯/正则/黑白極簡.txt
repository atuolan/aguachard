// 确保页面加载后执行标签生成
document.addEventListener('DOMContentLoaded', function() {
  const tagsContainer = document.getElementById('tags-container');
  
  if (tagsContainer) {
    // 获取现有的标签文本内容
    const existingTag = tagsContainer.querySelector('.tag');
    if (existingTag) {
      const tagText = existingTag.textContent;
      
      // 只有当内容包含分隔符时才进行分割
      if (tagText.includes('、')) {
        const tagItems = tagText.split('、');
        
        // 清空容器并重新创建标签
        tagsContainer.innerHTML = '';
        tagItems.forEach(item => {
          if (item.trim()) {
            const tagItem = document.createElement('span');
            tagItem.className = 'tag';
            tagItem.innerHTML = `<span>${item.trim()}</span>`;
            tagsContainer.appendChild(tagItem);
          }
        });
      } else {
        // 为单个标签添加span包装
        existingTag.innerHTML = `<span>${existingTag.textContent}</span>`;
      }
    }
  }
});
</script>
