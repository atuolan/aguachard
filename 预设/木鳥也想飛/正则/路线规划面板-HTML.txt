    <script>
      (function () {
        var currentScript = document.currentScript;
        if (!currentScript) {
          var scripts = document.getElementsByTagName('script');
          currentScript = scripts[scripts.length - 1];
        }

        var wrapper = currentScript.closest('.intersection-wrapper');
        var rawDataEl = wrapper.querySelector('.intersection-raw-data');
        var container = wrapper.querySelector('.intersection-options');

        if (rawDataEl && container) {
          var rawText = rawDataEl.textContent.trim();
          
          // 定义五个类别及其匹配正则
          var categories = [
            { name: '時間跳轉', regex: /fork[１1][:：]\s*(.*?)(?=(?:fork[２2]|$))/is, icon: '⏰' },
            { name: '場景發展', regex: /fork[２2][:：]\s*(.*?)(?=(?:fork[３3]|$))/is, icon: '🎬' },
            { name: 'NPC行動', regex: /fork[３3][:：]\s*(.*?)(?=(?:fork[４4]|$))/is, icon: '👤' },
            { name: '阻力形成', regex: /fork[４4][:：]\s*(.*?)(?=(?:fork[５5]|$))/is, icon: '⚠️' },
            { name: '機會形成', regex: /fork[５5][:：]\s*(.*?)(?=(?:$))/is, icon: '✨' }
          ];

          container.innerHTML = '';
          var foundCount = 0;

          categories.forEach(function(cat) {
            var match = rawText.match(cat.regex);
            if (match && match[1]) {
              var text = match[1].trim();
              if (text) {
                foundCount++;
                
                var option = document.createElement('div');
                option.className = 'intersection-option';
                
                var content = document.createElement('div');
                content.className = 'intersection-option-content';
                
                var category = document.createElement('div');
                category.className = 'intersection-option-category';
                category.textContent = cat.icon + ' 【' + cat.name + '】';
                
                var textEl = document.createElement('div');
                textEl.className = 'intersection-option-text';
                textEl.textContent = text;
                
                content.appendChild(category);
                content.appendChild(textEl);
                option.appendChild(content);
                
                // 点击事件处理
                (function(selectedText) {
                  option.onclick = function() {
                    if (typeof triggerSlash === 'function') {
                      triggerSlash('/setinput ' + selectedText);
                    } else {
                      console.log('Selected:', selectedText);
                    }
                  };
                })(text);
                
                container.appendChild(option);
              }
            }
          });

          if (foundCount === 0) {
            container.innerHTML = '<div class="empty-state">未找到有效的劇情分支選項</div>';
          }
        }
      })();
    </script>
  </div>
